	--no-vis,False (Aus),"Ein ""Schalter"". Wenn du ihn hinschreibst, wird das Visualisierungs-Fenster am Ende NICHT geöffnet. Nützlich für automatisierte Abläufe."

----PCD in Isaacsim----
	window->script editor:
import open3d as o3d
import numpy as np
from pxr import Usd, UsdGeom, Gf
import omni.usd

# --- KONFIGURATION ---
# Pfad zu deiner .pcd Datei (nutze / statt \ auch unter Windows/Linux)
pcd_path = "/home/bauya/map_saves/HomeLab_test3_OCTO.pcd" 
# Wo soll die Punktwolke in der Stage erscheinen?
prim_path = "/World/PointCloud"

try:
    # 1. PCD Datei einlesen
    print(f"Lade Datei: {pcd_path} ...")
    pcd = o3d.io.read_point_cloud(pcd_path)
    
    if not pcd.has_points():
        print("FEHLER: Die Datei ist leer oder konnte nicht gelesen werden.")
    else:
        # Konvertierung zu Numpy Arrays
        points = np.asarray(pcd.points)
        colors = np.asarray(pcd.colors)
        
        print(f"Erstelle Punktwolke mit {len(points)} Punkten...")

        # 2. Zugriff auf die aktuelle Stage
        stage = omni.usd.get_context().get_stage()
        
        # 3. USD PointCloud Prim definieren
        pc_prim = UsdGeom.Points.Define(stage, prim_path)
        
        # 4. Punkte (Positionen) setzen
        # USD erwartet Vt.Vec3fArray, aber Isaac Sim akzeptiert oft direkt Numpy Arrays
        pc_prim.CreatePointsAttr(points)
        
        # 5. Farben setzen (falls vorhanden)
        if len(colors) > 0:
            pc_prim.CreateDisplayColorAttr(colors)
        else:
            # Falls keine Farben da sind, mache sie weiß
            pc_prim.CreateDisplayColorAttr([(1, 1, 1)] * len(points))

        # 6. Breite der Punkte setzen (damit man sie sieht)
        pc_prim.CreateWidthsAttr([0.02] * len(points)) # 2cm Dicke pro Punkt

        print("Fertig! Schau in den Viewport.")

except Exception as e:
    print(f"Ein Fehler ist aufgetreten: {e}")
